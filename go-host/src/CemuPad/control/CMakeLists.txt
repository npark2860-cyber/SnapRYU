if(NOT TARGET CemuPadCore)
    add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../core ${CMAKE_CURRENT_BINARY_DIR}/core)
endif()
if(NOT TARGET CemuPadH264)
    add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../media ${CMAKE_CURRENT_BINARY_DIR}/media)
endif()
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL 3.5.9 EXACT REQUIRED COMPONENTS SSL Crypto)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/../third_party/nlohmann/json.hpp" JSON_HEADER_SHA)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/../third_party/nlohmann/LICENSE.MIT" JSON_LICENSE_SHA)
if(NOT JSON_HEADER_SHA STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63" OR
   NOT JSON_LICENSE_SHA STREQUAL "46a65cffd1ea955132d95a8dd921640714a8d6b537d2e4e482d31145ae95b603")
    message(FATAL_ERROR "Locked control JSON header/license changed")
endif()
set(CEMUPAD_CONTROL_SCHEMA_FILE "${CMAKE_CURRENT_LIST_DIR}/../../../../../protocol/control.schema.json" CACHE FILEPATH "Frozen control contract")
file(READ "${CEMUPAD_CONTROL_SCHEMA_FILE}" CEMUPAD_CONTROL_SCHEMA)
configure_file(ControlSchema.h.in ${CMAKE_CURRENT_BINARY_DIR}/generated/ControlSchema.h @ONLY)
add_library(CemuPadControl STATIC ControlMessage.cpp ControlSession.cpp PairingAuthority.cpp VerifiedTls.cpp ../crypto/AuthPrimitives.cpp)
set_target_properties(CemuPadControl PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(CemuPadControl PUBLIC ${CMAKE_CURRENT_LIST_DIR}/../.. PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_compile_features(CemuPadControl PUBLIC cxx_std_20)
target_link_libraries(CemuPadControl PUBLIC CemuPadCore CemuPadH264 OpenSSL::SSL OpenSSL::Crypto)
if(MSVC)
    target_compile_definitions(CemuPadControl PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(CemuPadControl PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(CemuPadControl PRIVATE -Wall -Wextra -Werror)
endif()
