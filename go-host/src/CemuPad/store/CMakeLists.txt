if(NOT WIN32)
    message(FATAL_ERROR "CemuPadStoreWindows requires Windows")
endif()
if(NOT TARGET CemuPadControl)
    add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../control ${CMAKE_CURRENT_BINARY_DIR}/control)
endif()
add_library(CemuPadStoreWindows STATIC Identity.cpp WindowsStore.cpp)
target_include_directories(CemuPadStoreWindows PUBLIC ${CMAKE_CURRENT_LIST_DIR}/../..)
target_compile_features(CemuPadStoreWindows PUBLIC cxx_std_20)
target_compile_definitions(CemuPadStoreWindows PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(CemuPadStoreWindows PUBLIC CemuPadControl PRIVATE crypt32 advapi32)
if(MSVC)
    target_compile_options(CemuPadStoreWindows PRIVATE /W4 /permissive- /utf-8)
endif()
